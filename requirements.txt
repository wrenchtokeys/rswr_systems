# Payments
# Unpinned major upgrades have broken webhooks before (v15 removed dict
# inheritance from StripeObject) — bump the ceiling deliberately, with tests.
#
# The floor matters as much as the ceiling: `>=8.0.0` let production resolve
# to 15.4.0 while dev machines sat on 14.3.0, and the two SDKs default to
# different Stripe API versions (dahlia vs clover), so payload shapes differed
# between the environment being tested and the one running. The API version is
# now pinned explicitly in settings (STRIPE_API_VERSION), but keep the SDK
# range narrow so local and prod agree by default.
stripe>=15.4.0,<16

# Supplier web-service calls (apps/technician_portal/mygrant_service.py).
# Already present transitively via stripe, but we import it directly now.
requests>=2.32.0,<3

# Parallel test runs (`manage.py test --parallel`). Django ships each worker's
# result back to the parent process by pickle, and a traceback object cannot be
# pickled — so without tblib the ENTIRE run dies with "TypeError: cannot pickle
# 'traceback' object" the moment any single test fails, printing no results at
# all. Django's own crash output names this package as the fix. Measured on a
# 16-core laptop: the full suite goes from 80 minutes serial to 20 minutes at
# --parallel 8. Test-only, but there is no requirements-dev.txt to put it in;
# it is pure Python and imports nothing at runtime.
tblib>=3.0
